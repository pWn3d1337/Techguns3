package techguns.devdump;

import java.io.File;
import java.io.FileWriter;
import java.io.Writer;
import java.lang.reflect.Array;
import java.lang.reflect.Field;
import java.lang.reflect.Modifier;
import java.util.ArrayList;
import java.util.Collection;
import java.util.IdentityHashMap;
import java.util.List;
import java.util.Map;

import com.google.gson.GsonBuilder;
import com.google.gson.JsonArray;
import com.google.gson.JsonObject;

import net.minecraftforge.client.event.GuiOpenEvent;
import net.minecraftforge.fml.common.Mod;
import net.minecraftforge.fml.common.eventhandler.SubscribeEvent;
import net.minecraftforge.fml.relauncher.Side;
import techguns.util.MBlock;
import techguns.world.structures.MBlockRegister;

/**
 * TEMPORARY dev tool for the Techguns 3 port, not part of Techguns 2: when the dev client opens its first screen it
 * writes run/role_dump.json, the {@code MBlockRegister} fields (the blocks Techguns 2 swaps for Chisel blocks when
 * Chisel is installed) that each structure palette entry is, by identity. Paths: the palette index, then for nested
 * MBlocks the field name and index ({@code 12/mBlocks/1}). Also lists MBlockRegister fields sharing one object.
 */
@Mod.EventBusSubscriber(modid = "techguns", value = Side.CLIENT)
public final class RoleDump {
	private static boolean done = false;

	@SubscribeEvent
	public static void onGuiOpen(GuiOpenEvent event) {
		if (done) {
			return;
		}
		done = true;
		try {
			Map<Object, List<String>> roles = new IdentityHashMap<>();
			for (Field f : MBlockRegister.class.getDeclaredFields()) {
				if (Modifier.isStatic(f.getModifiers()) && MBlock.class.isAssignableFrom(f.getType())) {
					f.setAccessible(true);
					Object v = f.get(null);
					if (v != null) {
						roles.computeIfAbsent(v, k -> new ArrayList<>()).add(f.getName());
					}
				}
			}
			JsonObject root = new JsonObject();
			JsonObject shared = new JsonObject();
			for (List<String> names : roles.values()) {
				if (names.size() > 1) {
					JsonArray arr = new JsonArray();
					names.forEach(arr::add);
					shared.add(names.get(0), arr);
				}
			}
			root.add("shared", shared);
			File dir = new File("../src/main/java/techguns/world/structures");
			for (File file : dir.listFiles()) {
				String n = file.getName();
				if (!n.endsWith(".java")) {
					continue;
				}
				Class<?> clazz = Class.forName("techguns.world.structures." + n.substring(0, n.length() - 5));
				JsonObject entry = new JsonObject();
				for (Field f : clazz.getDeclaredFields()) {
					if (!Modifier.isStatic(f.getModifiers()) || clazz == MBlockRegister.class) {
						continue;
					}
					f.setAccessible(true);
					Object value = f.get(null);
					JsonObject found = new JsonObject();
					walk(value, "", roles, found, new IdentityHashMap<>(), 0);
					if (found.size() > 0) {
						entry.add(f.getName(), found);
					}
				}
				if (entry.size() > 0) {
					root.add(clazz.getName(), entry);
				}
			}
			File out = new File("role_dump.json");
			try (Writer w = new FileWriter(out)) {
				new GsonBuilder().setPrettyPrinting().create().toJson(root, w);
			}
			System.out.println("TG ROLE DUMP DONE: " + out.getAbsolutePath());
		} catch (Throwable t) {
			System.out.println("TG ROLE DUMP FAILED");
			t.printStackTrace();
		}
	}

	private static void walk(Object v, String path, Map<Object, List<String>> roles, JsonObject found, Map<Object, Boolean> seen, int depth)
			throws Exception {
		if (v == null || depth > 8) {
			return;
		}
		if (v.getClass().isArray() && !v.getClass().getComponentType().isPrimitive()) {
			for (int i = 0; i < Array.getLength(v); i++) {
				walk(Array.get(v, i), join(path, Integer.toString(i)), roles, found, seen, depth + 1);
			}
			return;
		}
		if (v instanceof Collection) {
			int i = 0;
			for (Object o : (Collection<?>) v) {
				walk(o, join(path, Integer.toString(i++)), roles, found, seen, depth + 1);
			}
			return;
		}
		if (!(v instanceof MBlock) || seen.containsKey(v)) {
			return;
		}
		seen.put(v, true);
		List<String> names = roles.get(v);
		if (names != null && !path.isEmpty()) {
			JsonArray arr = new JsonArray();
			names.forEach(arr::add);
			found.add(path, arr);
		}
		for (Class<?> c = v.getClass(); c != Object.class; c = c.getSuperclass()) {
			for (Field f : c.getDeclaredFields()) {
				if (Modifier.isStatic(f.getModifiers())) {
					continue;
				}
				f.setAccessible(true);
				Object child = f.get(v);
				if (child instanceof MBlock || child instanceof Collection || (child != null && child.getClass().isArray())) {
					walk(child, join(path, f.getName()), roles, found, seen, depth + 1);
				}
			}
		}
		seen.remove(v);
	}

	private static String join(String path, String part) {
		return path.isEmpty() ? part : path + "/" + part;
	}
}
